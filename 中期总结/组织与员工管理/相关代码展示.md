```sql
-- 部门表：支持树形结构和多级管理

CREATE TABLE IF NOT EXISTS `sys_department` (

  `id` int NOT NULL AUTO_INCREMENT,

  `dname` varchar(50) NOT NULL COMMENT '部门名称',

  `dtype` varchar(20) DEFAULT NULL COMMENT '部门类型',

  `parent_id` int DEFAULT NULL COMMENT '父部门ID',

  `order_num` int DEFAULT NULL COMMENT '排序',

  `leader_id` int DEFAULT NULL COMMENT '部门负责人ID',

  PRIMARY KEY (`id`)

) ENGINE=InnoDB AUTO_INCREMENT=6 DEFAULT CHARSET=utf8mb4 COMMENT='系统部门表';



-- 岗位表：与部门关联，支持岗位级别和技能要求

CREATE TABLE IF NOT EXISTS `sys_post` (

  `id` int NOT NULL AUTO_INCREMENT,

  `pname` varchar(50) NOT NULL COMMENT '岗位名称',

  `ptype` varchar(20) DEFAULT NULL COMMENT '岗位类型',

  `organization` int DEFAULT NULL COMMENT '所属部门',

  `post_level` varchar(10) DEFAULT NULL COMMENT '岗位级别',

  `requirements` varchar(200) DEFAULT NULL COMMENT '岗位要求',

  PRIMARY KEY (`id`)

) ENGINE=InnoDB AUTO_INCREMENT=9 DEFAULT CHARSET=utf8mb4 COMMENT='系统岗位表';

```

#### 2.2 前端录取功能实现 (`PeroidOp.vue:70-90`)

```javascript

// 录取功能实现 - 自动设置试用期结束日期

handleAdmission(index, row) {

  this.$confirm('确定要录取该员工吗？此操作将设置试用期结束日期为当前时间', '提示', {

    confirmButtonText: '确定',

    cancelButtonText: '取消',

    type: 'warning'

  }).then(() => {

    // 获取当前日期作为试用期结束日期

    const currentDate = new Date();

    const endDate = currentDate.toISOString().split('T')[0]; // 格式化为 YYYY-MM-DD

    

    // 调用专门的接口来设置试用期结束日期

    this.$axios

      .post('/setEndDate?id=' + row.id +"&enddate=" + endDate)

      .then(successResponse => {

        if (successResponse.data === "ok") {

          this.$alert('录取成功，试用期结束日期已设置为当前日期', {confirmButtonText: '确定' })

          this.loadPeriods()

        }else {

          this.$alert('录取失败！', {confirmButtonText: '确定' })

        }

      })

  }).catch(() => {

    this.$message({

      type: 'info',

      message: '已取消录取'

    });

  });

}

```
